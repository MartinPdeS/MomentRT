14class SP3: 

15 """Solve two coupled simplified P3 moments using tetrahedral elements. 

16 

17 Parameters 

18 ---------- 

19 boundaries : mapping of str to RobinBoundary or ReflectingBoundary 

20 Conditions on all six faces. Robin boundaries must have zero effective 

21 reflectance, corresponding to a matched-index vacuum boundary. 

22 

23 Notes 

24 ----- 

25 Each material supplies its phase function. Henyey–Greenstein materials 

26 require anisotropy; isotropic, Rayleigh, and tabulated functions provide 

27 their own first three normalized Legendre moments. 

28 Illumination is isotropic over the incoming hemisphere. Fluence is the 

29 zeroth moment, not the first composite unknown. The implementation assumes 

30 continuous composite moments across internal interfaces and rejects index 

31 jumps. SP3 is an approximation, and agreement with SP3 slab references does 

32 not establish accuracy against radiative transfer for a particular sensor. 

33 """

144 missing = [

148 and isinstance( 

149 material.phase_function, MomentRT.phase_function.HenyeyGreenstein

152 

153 if missing: 

154 raise ValueError( 

155 "SP3 requires anisotropy on every material using Henyey–Greenstein; missing for: " 

156 + ", ".join(missing)

158 

159 phase_moments = np.array([material._phase_moments() for material in materials]) 

160 

161 anisotropy = phase_moments[:, 1] 

162 

163 factors = np.array([material._transport_factors() for material in materials]) 

164 

165 mu_a = np.array( 

166 [material.absorption.to("1/m").magnitude for material in materials] 

167 ) 

168 

169 mu_s1 = np.array( 

170 [material.reduced_scattering.to("1/m").magnitude for material in materials] 

171 ) 

172 

173 mu_1 = mu_a + mu_s1 

174 

175 mu_2 = mu_a + mu_s1 * factors[:, 2] 

176 

177 mu_3 = mu_a + mu_s1 * factors[:, 3] 

178 

179 first = self.boundary_matrix.copy() 

180 

181 second = self.boundary_matrix * (7 / 12) 

182 

183 coupling = self.boundary_matrix * (-1 / 4) 

184 

185 for label, (stiffness, mass) in self.region_matrices.items(): 

186 first += stiffness / (3 * mu_1[label]) + mu_a[label] * mass 

187 

188 second += ( 

189 stiffness / (7 * mu_3[label]) 

190 + (4 * mu_a[label] + 5 * mu_2[label]) / 9 * mass 

191 ) 

192 

193 coupling += (-2 / 3) * mu_a[label] * mass 

194 

195 matrix = sparse.bmat([[first, coupling], [coupling, second]], format="csr") 

196 

197 power = source.optical_power.to("W").magnitude 

198 

199 drive = self.unit_load * power 

200 

201 load = np.concatenate((drive, -drive / 4)) 

202 

203 solution = sparse_linalg.spsolve(matrix, load) 

204

206 raise RuntimeError("The SP3 solve returned nonfinite moments.") 

207 

208 relative_residual = np.linalg.norm(matrix @ solution - load) / np.linalg.norm( 

209 load 

210 ) 

211 

212 if relative_residual > 1e-8: 212 ↛ 213line 212 didn't jump to line 213 because the condition on line 212 was never true

213 raise RuntimeError( 

214 f"SP3 linear residual is too large: {relative_residual:g}." 

215 )

217 u, v = np.split(solution, 2) 

218 

219 scalar = u - 2 * v / 3 

220 

221 geometry = self.geometry

250 mu_a[self.labels] 

251 * geometry.volumes 

252 * scalar[geometry.tetrahedra].mean(axis=1)

255 boundary_fields: dict[str, MomentRT.fields.BoundaryField] = {}

257 escaping = {} 

258 

259 for face, triangles in geometry.faces.items(): 

260 quadrature, barycentric, weights = geometry.quadrature[face] 

261 

262 if isinstance(self.boundaries[face], MomentRT.boundary.ReflectingBoundary): 

263 outgoing = np.zeros(weights.shape) 

264 else: 

265 u_face = np.einsum("qa,ta->tq", barycentric, u[triangles]) 

266 

267 v_face = np.einsum("qa,ta->tq", barycentric, v[triangles]) 

268 

269 incoming = ( 

270 self.unit_irradiance * power if face == self.source.face else 0 

271 ) 

272 

273 outgoing = u_face / 2 - v_face / 8 - incoming 

274 

275 escaping[face] = float(np.sum(outgoing * weights)) 

276 

277 boundary_fields[face] = MomentRT.fields.BoundaryField( 

278 face=face, 

279 positions=quadrature * MomentRT.units.ureg.meter, 

280 area_weights=weights * MomentRT.units.ureg.meter**2,

282 sample_origin=self.medium.origin, 

283 sample_size=self.medium.size, 

284 wavelength=source.wavelength, 

285 ) 

286 

287 flux_values = -np.einsum( 

288 "ti,tij->tj", u[geometry.tetrahedra], geometry.gradients 

289 ) / (3 * mu_1[self.labels, None]) 

290 

291 reused = self.solve_count > 0 

292 

293 self.solve_count += 1

295 return MomentRT.results.SimulationResult(

304 "composite_2": field(values=v, name="SP3 composite moment 2"),
